// WsGateway.afterInit(server: Server) 中的中间件注册（真实节选）
server.use(async (socket, next) => {
  try {
    // 1) 契约 §6：io('/ws', { auth: { token } }) 携带 access token
    const token = extractSocketToken(socket.handshake.auth);
    // 2) 校验签名 / audience / 过期，并回查用户状态（封禁也在此拒绝）
    const user = await this.wsAuth.authenticate(token);
    // 3) 挂到 socket 上，后续事件直接取 socket.data.user
    socket.data.user = user;
    next();
  } catch (error) {
    // 4) 失败 → connect_error；message 与 data.code 同为错误码
    //    AUTH_REQUIRED / TOKEN_EXPIRED / AUTH_INVALID / FORBIDDEN
    const code = isAppError(error) ? error.code : ErrorCode.AUTH_INVALID;
    this.logger.warn(`WS 握手失败: ${code}`);
    const failure = new Error(code) as Error & { data?: Record<string, unknown> };
    failure.data = { code };
    next(failure);
  }
});

## ② 发消息（幂等 + 持久化）— 重构示例

来源：`server/src/messages/messages.service.ts#send`（调用侧）+ `server/src/common/utils/idempotency.ts#sendIdempotently`（幂等核心）。
下面把两处合并成一个等价片段，**整段 ≤40 行**；真实源码的 `insert` 里还包括线程计数、附件、提及通知。

```ts
// 调用侧（真实源码 send() 的裁剪）
async send(user: AuthUser, channelId: string, dto: SendMessageDto) {
  const resolved = await this.channelAccess.assert(channelId, user, 'write'); // 成员/禁言/归档
  await this.rateLimiter.assertMessageSend(user.id);                          // 5/s + 60/min
  const gateway: IdempotentGateway<SendInput, InsertedMessage> = {
    insert: async (p) => this.prisma.withTransaction((tx) => this.insertMessage(tx, p)),
    findExisting: async (p) => this.findByClientMsgId(p),
    isDuplicateError: (e) => PrismaService.isUniqueViolation(e),              // P2002
  };
  const input = { channelId, senderId: user.id, clientMsgId: dto.clientMsgId, dto, user, resolved };
  const { message, dup } = await sendIdempotently(gateway, input);
  if (!dup) this.realtime.emitToChannel(channelId, 'message:new', { message: message.view });
  return { message: message.view, dup }; // dup=true 时只回 ack，不广播（契约 §6）
}

// 幂等核心（真实源码 sendIdempotently 全文主体）
export async function sendIdempotently<TInput extends IdempotentInput, TMessage>(
  gateway: IdempotentGateway<TInput, TMessage>,
  input: TInput,
): Promise<{ message: TMessage; dup: boolean }> {
  try {
    const message = await gateway.insert(input);
    return { message, dup: false };
  } catch (error) {
    if (!gateway.isDuplicateError(error)) throw error; // 只吞唯一键冲突
  }
  const existing = await gateway.findExisting(input);
  if (!existing) {
    throw new AppError(ErrorCode.INTERNAL, '重复信号未能读取，请用同一 clientMsgId 重试');
  }
  return { message: existing, dup: true };
}
```

// afterInit(server: Server)（真实节选）
afterInit(server: Server): void {
  // REST 侧（消息/反应/卡片）与 WS 侧共用同一个 server 实例出口
  this.realtime.attach(server);

  // ioredis 的 pub/sub 双连接；未配置 REDIS_URL 时为 null（单实例内存 adapter）
  const pair = this.redis.adapterPair();
  if (pair) {
    // 挂上 adapter 后，任意实例 emit 都会经 Redis fan-out 到全部实例（多实例广播）
    server.adapter(createAdapter(pair.pubClient, pair.subClient));
    this.logger.log('Socket.IO Redis adapter 已启用（多实例广播）');
  } else {
    this.logger.warn('未配置 Redis：Socket.IO 使用单实例内存 adapter');
  }

  // /rtc 预留命名空间：恒 connect_error + NOT_IMPLEMENTED（契约 §11）

}

## ⑤ 已读回执节流（服务端 300ms 合并）— 真实源码节选

来源：`server/src/common/utils/read-throttle.ts` → `ReadReceiptMerger.submit()` / `flushKey()`

```ts
// 窗口内只保留最大 seq；窗口到期合并落库（真实节选）
submit(userId: string, channelId: string, seq: bigint): void {
  if (this.closed) return;
  const key = `${userId}:${channelId}`;
  const previous = this.pending.get(key);
  if (previous !== undefined && seq <= previous) return; // 契约 §3：已读只前进
  this.pending.set(key, seq);
  if (this.timers.has(key)) return;                       // 已开窗，等合并

  const timer = setTimeout(() => {
    this.timers.delete(key);
    void this.flushKey(key);
  }, this.windowMs);                                      // 契约 §9：300ms
  timer.unref?.();
  this.timers.set(key, timer);
}

private async flushKey(key: string): Promise<void> {
  const seq = this.pending.get(key);
  if (seq === undefined) return;
  this.pending.delete(key);
  const separator = key.indexOf(':');
  const entry = { userId: key.slice(0, separator), channelId: key.slice(separator + 1), seq };
  // ReadReceiptService：UPDATE ... WHERE last_read_seq < seq，成功后广播 receipt:update
  await this.flushFn(entry);
}
```

// FilesService.presign()（真实节选；错误处理略）
async presign(user: AuthUser, dto: PresignDto) {
  await this.channelAccess.assert(dto.channelId, user, 'write'); // 成员 + 未禁言 + 未归档

  const mime = normalizeMime(dto.mime);
  const policy = checkUpload(mime, dto.size);                    // 契约 §9 白名单 + 大小上限
  if (!policy.ok) throw new AppError(policy.code, policy.message);

  const storageKey = this.buildStorageKey(dto.channelId, dto.filename); // channels/<id>/<yyyy>/<mm>/<uuid>.<ext>
  const file = await this.prisma.file.create({                   // 先落 pending 行
    data: {
      ownerId: user.id, channelId: dto.channelId, bucket: this.s3.bucket, storageKey,
      originalName: dto.filename.slice(0, 255), mime, kind: policy.kind,
      sizeBytes: BigInt(dto.size), status: 'pending',
    },
  });

  const uploadUrl = await this.s3.presignPut(storageKey, mime);   // Content-Type 参与签名
  return {
    fileId: file.id, uploadUrl, method: 'PUT' as const,
    headers: { 'content-type': mime }, expiresIn: this.s3.presignExpiresSec,
  };
}
